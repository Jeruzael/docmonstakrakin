import {parentPort,workerData} from 'node:worker_threads';import{createRequire}from'node:module';import path from'node:path';import fs from'node:fs';import{performance}from'node:perf_hooks';
const require=createRequire(path.join(process.cwd(),'package.json'));const Database=require('better-sqlite3');const db=new Database(workerData.db,{timeout:100});db.pragma('journal_mode=WAL');db.pragma('synchronous=FULL');db.pragma('foreign_keys=ON');db.exec('CREATE TABLE IF NOT EXISTS bench(id INTEGER PRIMARY KEY, value TEXT NOT NULL, rev INTEGER NOT NULL DEFAULT 1);CREATE TABLE IF NOT EXISTS tokens(digest TEXT PRIMARY KEY,payload TEXT NOT NULL)');
const insert=db.prepare('INSERT INTO bench(id,value) VALUES(?,?)');db.transaction(()=>{for(let i=0;i<1000;i++)insert.run(i,'synthetic-'+i)})();const read=db.prepare('SELECT value,rev FROM bench WHERE id=?');const write=db.prepare('UPDATE bench SET value=?,rev=rev+1 WHERE id=?');
parentPort.postMessage({ready:true,sqlite:db.prepare('select sqlite_version() as v').get().v});
parentPort.on('message',async m=>{try{if(m.action==='roundtrip'){parentPort.postMessage({id:m.id,value:read.get(m.value%1000)});return;}
if(m.action==='measure'){const samples={read:[],write:[]};for(let i=0;i<2000;i++){let t=performance.now();read.get(i%1000);samples.read.push(performance.now()-t);}for(let i=0;i<80;i++){let t=performance.now();db.transaction(()=>write.run('measured',i))();samples.write.push(performance.now()-t);}parentPort.postMessage({id:m.id,samples});return;}
if(m.action==='failure'){const before=db.prepare('SELECT sum(rev) as n FROM bench').get().n;try{db.transaction(()=>{write.run('rolled-back',0);throw Error('synthetic rollback')})();}catch{}const rollback=db.prepare('SELECT sum(rev) as n FROM bench').get().n===before;let readonly=false,corrupt=false,busy=false;let t=performance.now();const other=new Database(workerData.db,{timeout:100});db.exec('BEGIN IMMEDIATE');try{other.prepare('UPDATE bench SET rev=rev+1 WHERE id=0').run();}catch(e){busy=e.code==='SQLITE_BUSY';}const busyMs=performance.now()-t;db.exec('ROLLBACK');other.close();const ro=new Database(workerData.db,{readonly:true});try{ro.prepare('UPDATE bench SET rev=rev+1 WHERE id=0').run();}catch(e){readonly=e.code==='SQLITE_READONLY';}ro.close();const bad=workerData.db+'.corrupt';fs.writeFileSync(bad,'not a sqlite file');try{const broken=new Database(bad);try{broken.pragma('quick_check');}finally{broken.close();}}catch{corrupt=true;}parentPort.postMessage({id:m.id,rollback,readonly,corrupt,busy,busyMs});return;}
if(m.action==='tokenPut'){db.prepare('INSERT INTO tokens(digest,payload) VALUES(?,?)').run(m.digest,JSON.stringify(m.payload));parentPort.postMessage({id:m.id});return;}
if(m.action==='tokenGet'){const r=db.prepare('SELECT payload FROM tokens WHERE digest=?').get(m.digest);parentPort.postMessage({id:m.id,payload:r?JSON.parse(r.payload):null});return;}
if(m.action==='tokenDelete'){db.prepare('DELETE FROM tokens WHERE digest=?').run(m.digest);parentPort.postMessage({id:m.id});return;}
if(m.action==='checkpoint'){db.pragma('wal_checkpoint(TRUNCATE)');parentPort.postMessage({id:m.id,integrity:db.pragma('quick_check',{simple:true})});return;}
if(m.action==='countCrash'){parentPort.postMessage({id:m.id,count:db.prepare('SELECT count(*) AS n FROM bench WHERE id=9999').get().n});return;}
if(m.action==='close'){db.close();parentPort.postMessage({id:m.id});parentPort.close();return;}throw Error('unknown probe command');
}catch(e){parentPort.postMessage({id:m.id,error:e.message});}});
